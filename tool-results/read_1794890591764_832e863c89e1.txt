   385→// and document understanding can exceed the shorter 60-second default.

   393→      return NextResponse.json(
   394→        { error: 'API Key Gemini diperlukan. Masukkan di pengaturan chat.' },
   395→        { status: 400 }
   396→      );
   397→    }
   398→
   399→    // Interactions API accepts multimodal content directly and is the supported
   400→    // API for current Gemini models.
   401→    const lastMessage = messages[messages.length - 1];
   402→    const incomingFiles: { name: string; type: string; base64: string }[] =
   403→      files || lastMessage.files || [];
   404→
   405→    const input: Array<Record<string, string>> = [];
   406→
   407→    for (const file of incomingFiles) {
   408→      try {
   409→        const processed = await processUploadedFile(file);
   410→        if (processed.type === 'inlineData' && processed.inlineData) {
   411→          const mimeType = processed.inlineData.mimeType;
   412→          input.push({
   413→            type: mimeType === 'application/pdf' ? 'document' : 'image',
   414→            data: processed.inlineData.data,
   415→            mime_type: mimeType,
   416→          });
   417→        } else if (processed.text) {
   418→          input.push({ type: 'text', text: `Isi lampiran ${file.name}:\n${processed.text}` });
   419→        }
   420→      } catch (fErr) {
   421→        console.warn(`Error processing file ${file.name}:`, fErr);
   422→      }
   423→    }
   424→
   425→    const userPrompt = lastMessage.content?.trim() || (incomingFiles.length > 0
   426→      ? 'Tolong telaah dan analisis dokumen/gambar yang dilampirkan untuk kasus perhitungan SAR.'
   427→      : 'Halo');

   429→    // make the model call slow, especially after repeated failed attempts).
   430→    const conversation = messages.slice(-13, -1)
   431→      .filter((msg: { role: string; content: string }) =>
   432→        !(msg.role !== 'user' && msg.content?.startsWith('❌ Error:'))
   433→      )
   434→      .map((msg: { role: string; content: string }) =>
   435→        `${msg.role === 'user' ? 'Pengguna' : 'Asisten'}: ${(msg.content || '').slice(0, 6000)}`
   436→      )
   437→      .join('\n');

   439→      .filter((part) => part.type === 'text')
   440→      .map((part) => part.text || '')
   441→      .join('\n');

   443→      type: 'text',

   449→    const interceptCaseDetected = /\b(intercept|intersep|pintasan|medevac)\b/i.test(interceptSourceText)

   453→    const normalizeSchema = (value: unknown): unknown => {
   454→      if (Array.isArray(value)) return value.map(normalizeSchema);
   455→      if (!value || typeof value !== 'object') return value;
   456→      return Object.fromEntries(Object.entries(value).map(([key, child]) => [
   457→        key,
   458→        key === 'type' && typeof child === 'string' ? child.toLowerCase() : normalizeSchema(child),
   459→      ]));
   460→    };
   461→    const geminiStartedAt = Date.now();
   462→    let result: Record<string, unknown> | undefined;
   463→    let lastError: Error | undefined;
   464→    let usedModel: (typeof GEMINI_MODELS)[number] = GEMINI_MODELS[0];
   465→    for (const model of GEMINI_MODELS) {
   466→      for (let attempt = 0; attempt <= MAX_RETRIES_PER_MODEL; attempt += 1) {
   467→        console.info(`[AI chat] Gemini request started (model=${model}, attempt=${attempt + 1}, files=${incomingFiles.length}).`);
   468→        try {
   469→          const response = await fetch('https://generativelanguage.googleapis.com/v1beta/interactions', {
   470→            method: 'POST',
   471→            headers: { 'Content-Type': 'application/json', 'x-goog-api-key': apiKey },
   472→            signal: AbortSignal.timeout(GEMINI_REQUEST_TIMEOUT_MS),

   475→              input,
   476→              system_instruction: SYSTEM_PROMPT,
   477→              generation_config: {
   478→                max_output_tokens: 8192,

   481→                  tool_choice: {
   482→                    allowed_tools: { mode: 'any', tools: ['set_intercept_case'] },
   483→                  },
   484→                } : {}),
   485→              },
   486→              tools: tools.map((tool) => ({
   487→                type: 'function',
   488→                name: tool.name,
   489→                description: tool.description,
   490→                parameters: normalizeSchema(tool.parameters),
   491→              })),
   492→              store: false,
   493→            }),

   499→            usedModel = model;

   507→        } catch (requestError) {

   509→          if (requestError instanceof Error && (requestError.name === 'TimeoutError' || requestError.name === 'AbortError')) {

   515→        }

   519→        }

   523→    }
   524→    if (!result) {

   530→    }
   531→
   532→    const steps = Array.isArray(result.steps) ? result.steps as Array<{ type?: string; name?: string; arguments?: unknown; content?: Array<{ type?: string; text?: string }> }> : [];
   533→    const functionCalls = steps.filter((step): step is typeof step & { name: string } =>
   534→      step.type === 'function_call' && typeof step.name === 'string'
   535→    );

   540→      }));

   542→      const worksheetCalls = calls.filter((call) => call.name !== 'ask_for_case_data');

   546→        if (questions.length === 0) return '';

   552→        .filter((part: { type?: string }) => part.type === 'text')
   553→        .map((part: { text?: string }) => part.text || '')
   554→        .join('');
   555→
   556→      const interceptCall = calls.find((call) => call.name === 'set_intercept_case');
   557→      if (interceptCaseDetected && !interceptCall) {
   558→        return NextResponse.json({
   559→          text: clarificationText || 'Agar saya bisa menghitung Intercept, mohon berikan posisi awal target dan SRU/pangkalan, haluan serta kecepatan target, dan kecepatan SRU. Waktu referensi diperlukan bila Anda ingin melihat ETA.',
   560→          functionCalls: worksheetCalls.length ? worksheetCalls : null,
   561→        });
   562→      }
   563→      const responseText = interceptCall
   564→        ? 'Data kasus Intercept berhasil dianalisis. Saya memasukkan data ke worksheet Intercept; hasil hitung dan vektornya akan dibuat oleh kalkulator aplikasi.'
   565→        : text || 'Data telah disiapkan. Klik "Terapkan" untuk mengisi worksheet.';
   566→      return NextResponse.json({
   567→        text: `${usedModel !== GEMINI_MODELS[0] ? 'Model utama sedang sibuk; jawaban diproses dengan model cadangan.\n\n' : ''}${responseText}${clarificationText}`,
   568→        functionCalls: worksheetCalls.length ? worksheetCalls : null,
   569→      });
   570→    }
   571→
   572→    if (interceptCaseDetected) {
   573→      return NextResponse.json({
   574→        text: 'Agar saya bisa menghitung Intercept, mohon berikan posisi awal target dan SRU/pangkalan, haluan serta kecepatan target, dan kecepatan SRU. Waktu referensi diperlukan bila Anda ingin melihat ETA.',
   575→        functionCalls: null,
   576→      });
   577→    }
   578→
   579→    const responseText = steps
   580→      .filter((step: { type?: string }) => step.type === 'model_output')
   581→      .flatMap((step: { content?: Array<{ type?: string; text?: string }> }) => step.content || [])
   582→      .filter((part: { type?: string }) => part.type === 'text')
   583→      .map((part: { text?: string }) => part.text || '')
   584→      .join('');
   585→    return NextResponse.json({
   586→      text: `${usedModel !== GEMINI_MODELS[0] ? 'Model utama sedang sibuk; jawaban diproses dengan model cadangan.\n\n' : ''}${responseText || 'Model Gemini tidak mengembalikan teks jawaban.'}`,
   587→      functionCalls: null,
   588→    });
   589→  } catch (error: unknown) {
   590→    const message = error instanceof Error ? error.message : 'Unknown error';
   591→    console.error('Chat API error:', message);
   592→    return NextResponse.json({ error: `Gagal menghubungi Gemini: ${message}` }, { status: 500 });
   593→  }
   594→}
