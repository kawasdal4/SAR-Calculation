   119→function formatDMS(value: number, isLat: boolean): string {
   120→  const hemi = isLat ? (value >= 0 ? 'N' : 'S') : value >= 0 ? 'E' : 'W';
   121→  const abs = Math.abs(value);
   122→  const deg = Math.floor(abs);
   123→  const minFull = (abs - deg) * 60;
   124→  const min = Math.floor(minFull);
   125→  const sec = ((minFull - min) * 60).toFixed(1);
   126→  const dStr = deg.toString().padStart(isLat ? 2 : 3, '0');
   127→  const mStr = min.toString().padStart(2, '0');
   128→  return `${dStr}°${mStr}′${sec.padStart(4, '0')}″${hemi}`;
   129→}

   147→interface ViewPrefs {

   155→const DEFAULT_VIEW_PREFS: ViewPrefs = {
   156→  layers: true,
   157→  coord: true,
   158→  zoom: true,
   159→  disclaimer: true,
   160→  depth: false, // off by default — opt-in to avoid surprise fetches

   163→const VIEW_PREFS_KEY = 'seamap:viewPrefs';
   164→const DEPTH_DENSITY_KEY = 'seamap:depthDensity';
   165→
   166→function loadViewPrefs(): ViewPrefs {
   167→  try {
   168→    if (typeof window === 'undefined') return { ...DEFAULT_VIEW_PREFS };
   169→    const raw = window.localStorage.getItem(VIEW_PREFS_KEY);
   170→    if (!raw) return { ...DEFAULT_VIEW_PREFS };
   171→    const parsed = JSON.parse(raw) as Partial<ViewPrefs>;
   172→    return {
   173→      layers: typeof parsed.layers === 'boolean' ? parsed.layers : true,
   174→      coord: typeof parsed.coord === 'boolean' ? parsed.coord : true,
   175→      zoom: typeof parsed.zoom === 'boolean' ? parsed.zoom : true,
   176→      disclaimer: typeof parsed.disclaimer === 'boolean' ? parsed.disclaimer : true,
   177→      depth: typeof parsed.depth === 'boolean' ? parsed.depth : false,
   178→    };
   179→  } catch {
   180→    return { ...DEFAULT_VIEW_PREFS };
   181→  }
   182→}
   183→
   184→function saveViewPrefs(prefs: ViewPrefs): void {
   185→  try {
   186→    if (typeof window === 'undefined') return;
   187→    window.localStorage.setItem(VIEW_PREFS_KEY, JSON.stringify(prefs));
   188→  } catch {
   189→    // ignore write failures (private mode, sandbox, etc.)
   190→  }
   191→}
   192→
   193→function allHidden(p: ViewPrefs): boolean {
   194→  // "Hide all" only hides the in-map overlays (layers/coord/zoom/disclaimer).
   195→  // The depth layer has its own checkbox in the Layers panel — it's not
   196→  // affected by the Tampilan "hide all" toggle.
   197→  return !p.layers && !p.coord && !p.zoom && !p.disclaimer;
   198→}
   199→
   200→function loadDepthDensity(): Density {
   201→  try {
   202→    if (typeof window === 'undefined') return 'sedang';
   203→    const raw = window.localStorage.getItem(DEPTH_DENSITY_KEY);
   204→    if (raw === 'rapat' || raw === 'sedang' || raw === 'jarang') return raw;
   205→  } catch {
   206→    // ignore
   207→  }
   208→  return 'sedang';
   209→}
   210→
   211→function saveDepthDensity(d: Density): void {
   212→  try {
   213→    if (typeof window === 'undefined') return;
   214→    window.localStorage.setItem(DEPTH_DENSITY_KEY, d);
   215→  } catch {
   216→    // ignore
   217→  }
   218→}

   368→
   369→    // ---- Nautical chart scale border (canvas-based) ----
   370→    // Style: NO black/white band. Each side is a 1 px outline at the very
   371→    // edge, with short tick marks projecting INTO the map area:
   372→    //   - minor tick: 5 px long, every `minorSec` arc-seconds
   373→    //   - major tick: 10 px long, every 30′ (or coarser at very far zoom)
   374→    // Ticks are dark (#1a1a1a) with a thin white halo (3 px stroke first,
   375→    // then 1 px dark on top) so they read on both the road map and satellite
   376→    // tiles. Labels (every major interval) sit just inside the map, with a
   377→    // white pill background.
   378→    //
   379→    // Label-interval contract:
   380→    //   - Major (labelled) interval is LOCKED to multiples of 30′
   381→    //     (1800″, 3600″, 7200″, 18000″ — i.e. 30′, 1°, 2°, 5°).
   382→    //   - Default major interval is 30′. Step up to 1°/2°/5° ONLY when 30′
   383→    //     labels would be too tight (per-axis threshold: 60 px for lng,
   384→    //     30 px for lat — lat threshold is lower so 30′ labels appear on
   385→    //     left/right even at wide zoom).
   386→    //   - At very close zoom, major interval stays at 30′; minor ticks
   387→    //     densify (1′, 30″, …) as long as they stay ≥ 5 px apart on screen.
   388→    //   - Labels are NEVER shown at 20′, 10′, 5′, 2′ or 1′.
   389→    //
   390→    // The four sides share one `drawBand` function — no per-side duplication.
   391→    const MAJOR_TICK_PX = 10;
   392→    const MINOR_TICK_PX = 5;
   393→    const LABEL_MIN_PX_LNG = 60;
   394→    const LABEL_MIN_PX_LAT = 30;
   395→    const MINOR_MIN_PX = 5;
   396→    const MINOR_DENSIFY_PX = 60;
   397→
   398→    const MAJOR_LABEL_STEPS_SEC = [1800, 3600, 7200, 18000];
   399→
   400→    let scaleBorderCanvas: HTMLCanvasElement | null = null;
   401→    let scaleBorderRAF: number | null = null;
   402→
   403→    function chooseMajorInterval(pxPerDeg: number, minLabelPx: number): number {
   404→      for (const step of MAJOR_LABEL_STEPS_SEC) {
   405→        if (step * pxPerDeg / 3600 >= minLabelPx) return step;
   406→      }
   407→      return MAJOR_LABEL_STEPS_SEC[MAJOR_LABEL_STEPS_SEC.length - 1];
   408→    }
   409→
   410→    function chooseMinorInterval(majorSec: number, pxPerDeg: number): number {
   411→      const spacing5 = 300 * pxPerDeg / 3600;
   412→      if (spacing5 >= MINOR_MIN_PX) {
   413→        if (spacing5 > MINOR_DENSIFY_PX) {
   414→          for (const c of [30, 60]) {
   415→            if (c * pxPerDeg / 3600 >= MINOR_MIN_PX && c < majorSec) return c;
   416→          }
   417→          return 300;
   418→        }
   419→        return 300;
   420→      }
   421→      for (const c of [600, 900]) {
   422→        if (c < majorSec && c * pxPerDeg / 3600 >= MINOR_MIN_PX) return c;
   423→      }
   424→      return majorSec;
   425→    }
   426→
   427→    /**
   428→     * Unified tick-drawing routine used by all four sides of the chart border.
   429→     * Top/bottom iterate along the longitude axis; left/right iterate along
   430→     * the latitude axis. The exact same tick + label rules apply to every
   431→     * side — only the orientation differs.
   432→     *
   433→     * No black/white band fill is drawn — only:
   434→     *   - tick marks (with white halo) projecting INTO the map from the edge,
   435→     *   - labels at major ticks (just inside the map, with black pill bg).
   436→     * The outer 1 px outline is drawn separately in drawScaleBorder().
   437→     *
   438→     * Labels are collected (not drawn immediately) so drawScaleBorder() can
   439→     * run a single bounding-box collision pass over ALL labels of ALL four
   440→     * sides together — a label that overlaps another label, or overlaps the
   441→     * scale/attribution controls, is dropped instead of stacked.
   442→     */
   443→    function drawBand(args: {
   444→      ctx: CanvasRenderingContext2D;
   445→      orientation: 'top' | 'bottom' | 'left' | 'right';
   446→      axis: CoordAxis;
   447→      axisStart: number;
   448→      axisEnd: number;
   449→      majorSec: number;
   450→      minorSec: number;
   451→      w: number;
   452→      h: number;
   453→      toScreenPos: (axisValueDeg: number) => number;
   454→      // Collect a label candidate (returns the bounding box that would be
   455→      // drawn, or null if the label would not fit inside the map area).
   456→      collectLabel: (text: string, x: number, y: number, align: 'center' | 'left' | 'right', baseline: 'top' | 'middle' | 'bottom', side: 'top' | 'bottom' | 'left' | 'right') => void;
   457→    }): void {
   458→      const {
   459→        ctx, orientation, axis, axisStart, axisEnd,
   460→        majorSec, minorSec, w, h,
   461→        toScreenPos, collectLabel,
   462→      } = args;
   463→
   464→      const isHorizontal = orientation === 'top' || orientation === 'bottom';
   465→      const viewportSpan = isHorizontal ? w : h;
   466→      const majSec = Math.round(majorSec);
   467→      const minSec = Math.round(minorSec);
   468→
   469→      // Iterate in integer arc-seconds to avoid floating-point drift.
   470→      // For latitude, axisStart = south (most negative), axisEnd = north
   471→      // (least negative / positive). startSec ≤ endSec because ceil() of a
   472→      // more-negative number divided by a positive step yields a smaller
   473→      // (more negative) integer.
   474→      const startSec = Math.ceil(axisStart * 3600 / minSec) * minSec;
   475→      const endSec = Math.ceil(axisEnd * 3600 / minSec) * minSec;
   476→
   477→      const LABEL_OFFSET = MAJOR_TICK_PX + 4;
   478→
   479→      for (let sec = startSec; sec <= endSec; sec += minSec) {
   480→        const axisValue = sec / 3600;
   481→        const wrappedAxisValue = axis === 'lng'
   482→          ? (((axisValue + 540) % 360) + 360) % 360 - 180
   483→          : axisValue;
   484→        const pos = toScreenPos(wrappedAxisValue);
   485→
   486→        if (pos < -40 || pos > viewportSpan + 40) continue;
   487→
   488→        const isMajor = (sec % majSec) === 0;
   489→        const tickLen = isMajor ? MAJOR_TICK_PX : MINOR_TICK_PX;
   490→
   491→        let x0: number, y0: number, x1: number, y1: number;
   492→        if (orientation === 'top') {
   493→          x0 = pos; y0 = 0; x1 = pos; y1 = tickLen;
   494→        } else if (orientation === 'bottom') {
   495→          x0 = pos; y0 = h - tickLen; x1 = pos; y1 = h;
   496→        } else if (orientation === 'left') {
   497→          x0 = 0; y0 = pos; x1 = tickLen; y1 = pos;
   498→        } else {
   499→          x0 = w - tickLen; y0 = pos; x1 = w; y1 = pos;
   500→        }
   501→
   502→        // Halo first (white, 3 px, round caps).
   503→        ctx.lineCap = 'round';
   504→        ctx.strokeStyle = 'rgba(255,255,255,0.95)';
   505→        ctx.lineWidth = 3;
   506→        ctx.beginPath();
   507→        ctx.moveTo(x0, y0);
   508→        ctx.lineTo(x1, y1);
   509→        ctx.stroke();
   510→
   511→        // Then the dark tick on top (1 px).
   512→        ctx.strokeStyle = '#1a1a1a';
   513→        ctx.lineWidth = 1;
   514→        ctx.beginPath();
   515→        ctx.moveTo(x0, y0);
   516→        ctx.lineTo(x1, y1);
   517→        ctx.stroke();
   518→
   519→        // Collect label candidate at major ticks. Spacing/collision is
   520→        // resolved later by the caller — here we just collect every
   521→        // candidate whose tick position is inside the visible area (with a
   522→        // 14 px / 20 px margin so the label has room near the corners).
   523→        if (isMajor) {
   524→          const label = formatDegMinSec(wrappedAxisValue, axis, majorSec);
   525→          if (isHorizontal) {
   526→            if (orientation === 'top') {
   527→              if (pos > MAJOR_TICK_PX + 20 && pos < w - MAJOR_TICK_PX - 20) {
   528→                collectLabel(label, pos, LABEL_OFFSET, 'center', 'top', 'top');
   529→              }
   530→            } else {
   531→              if (pos > MAJOR_TICK_PX + 20 && pos < w - MAJOR_TICK_PX - 20) {
   532→                collectLabel(label, pos, h - LABEL_OFFSET, 'center', 'bottom', 'bottom');
   533→              }
   534→            }
   535→          } else {

   586→        lat: centerPt.lat,

   603→      const eastLng = bounds.getEast();

   605→      const northLat = Math.min(85, bounds.getNorth());

   607→      const font = '11px "Courier New", ui-monospace, monospace';

   609→      const padY = 2;   // vertical padding inside the label pill
   610→      const th = 14 + padY * 2; // text height + padding = bg height
   611→
   612→      // Measure helper: returns bounding rect for a label at (x, y) with the
   613→      // given alignment/baseline. The pill width follows the text width
   614→      // (ctx.measureText) — no fixed width.
   615→      function measureLabel(
   616→        text: string,
   617→        x: number,
   618→        y: number,
   619→        align: 'center' | 'left' | 'right',
   620→        baseline: 'top' | 'middle' | 'bottom',
   621→      ): { bgX: number; bgY: number; tw: number; th: number } {
   622→        if (!ctx) return { bgX: 0, bgY: 0, tw: 0, th: 0 };
   623→        ctx.font = font;
   624→        const tw = ctx.measureText(text).width + padX * 2;
   625→        let bgX = x - tw / 2;
   626→        if (align === 'left') bgX = x - padX;
   627→        if (align === 'right') bgX = x - tw + padX;
   628→        const bgY = baseline === 'top' ? y - padY : baseline === 'middle' ? y - th / 2 : y - th + padY;
   629→        return { bgX, bgY, tw, th };
   630→      }
   631→
   632→      // Collect candidates from all four sides first. drawBand pushes into
   633→      // this array via the collectLabel closure.
   634→      const candidates: LabelCandidate[] = [];

   636→      function collectLabel(
   637→        text: string,
   638→        x: number,
   639→        y: number,
   640→        align: 'center' | 'left' | 'right',
   641→        baseline: 'top' | 'middle' | 'bottom',
   642→        side: 'top' | 'bottom' | 'left' | 'right',
   643→      ) {
   644→        const { bgX, bgY, tw, th: ths } = measureLabel(text, x, y, align, baseline);
   645→        // Reject candidates whose bounding box would extend past the map
   646→        // edges (would overlap the perpendicular border).
   647→        if (bgX < 1 || bgX + tw > w - 1 || bgY < 1 || bgY + ths > h - 1) return;
   648→        candidates.push({ text, x, y, align, baseline, side, bgX, bgY, tw, th: ths });
   649→      }
   650→
   651→      drawBand({
   652→        ctx, orientation: 'top', axis: 'lng',
   653→        axisStart: westLng, axisEnd: eastLng,
   654→        majorSec: lngMajSec, minorSec: lngMinSec,
   655→        w, h,
   656→        toScreenPos: (lng: number) =>
   657→          map.latLngToContainerPoint({ lat: 0, lng } as L.LatLngExpression).x,
   658→        collectLabel,
   659→      });
   660→      drawBand({
   661→        ctx, orientation: 'bottom', axis: 'lng',
   662→        axisStart: westLng, axisEnd: eastLng,
   663→        majorSec: lngMajSec, minorSec: lngMinSec,

   667→        collectLabel,
   668→      });
   669→      drawBand({
   670→        ctx, orientation: 'left', axis: 'lat',
   671→        axisStart: southLat, axisEnd: northLat,
   672→        majorSec: latMajSec, minorSec: latMinSec,
   673→        w, h,

   675→          map.latLngToContainerPoint({ lat, lng: 0 } as L.LatLngExpression).y,
   676→        collectLabel,
   677→      });
   678→      drawBand({
   679→        ctx, orientation: 'right', axis: 'lat',
   680→        axisStart: southLat, axisEnd: northLat,
   681→        majorSec: latMajSec, minorSec: latMinSec,
   682→        w, h,
   683→        toScreenPos: (lat: number) =>
   684→          map.latLngToContainerPoint({ lat, lng: 0 } as L.LatLngExpression).y,
   685→        collectLabel,
   686→      });
   687→
   688→      // ---- Collision resolution ----
   689→      // Reserved rectangles: the Leaflet scale control (bottom-left) and
   690→      // attribution (bottom-right) live inside the map. We don't want labels
   691→      // to overlap them. The actual control sizes vary, so reserve a generous
   692→      // strip in the bottom-left and bottom-right corners.
   693→      const reservedRects: Array<{ x: number; y: number; w: number; h: number }> = [
   694→        // Bottom-left: scale control (metric + nautical ≈ 110×34 px).
   695→        { x: 0, y: h - 40, w: 130, h: 40 },
   696→        // Bottom-right: attribution (text varies; reserve 220×24 px).
   697→        { x: w - 220, y: h - 28, w: 220, h: 28 },
   698→      ];
   699→
   700→      // Sort candidates by priority: top/bottom first (they're spread along
   701→      // the longer edge), then left/right. Within a side, keep iteration
   702→      // order (which is ascending latitude for left/right, ascending
   703→      // longitude for top/bottom). Stable sort preserves insertion order
   704→      // for ties.
   705→      const sideOrder: Record<string, number> = { top: 0, bottom: 1, left: 2, right: 3 };
   706→      candidates.sort((a, b) => sideOrder[a.side] - sideOrder[b.side]);
   707→
   708→      function rectsOverlap(
   709→        a: { x: number; y: number; w: number; h: number },
   710→        b: { x: number; y: number; w: number; h: number },
   711→      ): boolean {
   712→        return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
   713→      }
   714→
   715→      const placedRects: Array<{ x: number; y: number; w: number; h: number }> = [...reservedRects];
   716→      const kept: LabelCandidate[] = [];
   717→      for (const cand of candidates) {
   718→        const rect = { x: cand.bgX, y: cand.bgY, w: cand.tw, h: cand.th };
   719→        let collides = false;
   720→        for (const placed of placedRects) {
   721→          if (rectsOverlap(rect, placed)) { collides = true; break; }
   722→        }
   723→        if (collides) continue;
   724→        kept.push(cand);
   725→        placedRects.push(rect);
   726→      }
   727→
   728→      // ---- Draw kept labels ----
   729→      // Style: black bg (#000, opacity 0.85), white text, no outline,
   730→      // padding 4 px / 2 px, border-radius 2 px, font 11 px monospace.
   731→      // Sized to text via measureText (already done in measureLabel).
   732→      if (ctx) {
   733→        ctx.font = font;
   734→        ctx.textAlign = 'left';
   735→        ctx.textBaseline = 'top';
   736→        for (const cand of kept) {
   737→          // Background pill: black, opacity 0.85.
   738→          ctx.fillStyle = 'rgba(0,0,0,0.85)';
   739→          // Border-radius 2 px is approximated by drawing a rect (canvas v1
   740→          // supports roundRect, but the type defs in older @types/canvas
   741→          // don't — fall back to plain fillRect for portability).
   742→          if (typeof ctx.roundRect === 'function') {
   743→            ctx.beginPath();
   744→            ctx.roundRect(cand.bgX, cand.bgY, cand.tw, cand.th, 2);
   745→            ctx.fill();
   746→          } else {
   747→            ctx.fillRect(cand.bgX, cand.bgY, cand.tw, cand.th);
   748→          }
   749→          // Text: white, no outline. Position by textBaseline=text-top
   750→          // inside the pill.
   751→          ctx.fillStyle = '#ffffff';
   752→          // Place text inside the pill: left edge = bgX + padX, vertical
   753→          // center = bgY + th/2 + ~3 (font ascent correction for 11 px).
   754→          ctx.fillText(cand.text, cand.bgX + padX, cand.bgY + padY + 1);
   755→        }
   756→      }
   757→
   758→      // ===== Outer 1 px black outline around the whole map =====
   759→      if (ctx) {
   760→        ctx.strokeStyle = '#1a1a1a';
   761→        ctx.lineWidth = 1;
   762→        ctx.strokeRect(0.5, 0.5, w - 1, h - 1);
   763→      }
   764→
   765→      ctx?.restore();
   766→    }

   790→    // A separate canvas that sits ABOVE the tile panes but BELOW the

   796→    //   1. Computes the grid interval for the current zoom + density.
   797→    //   2. Enumerates grid points inside the viewport.

   799→    //   4. Draws every cached ocean point as a number with white halo.

   809→    function drawDepthOverlay() {
   810→      if (!depthCanvas) return;
   811→      // The depth canvas is hidden by CSS when the layer is disabled
   812→      // (`.seamap-hide-depth`), but we also bail here so we don't waste CPU
   813→      // drawing on a hidden canvas.
   814→      if (!depthEnabledRef.current) {
   815→        const ctx0 = depthCanvas.getContext('2d');
   816→        ctx0?.clearRect(0, 0, depthCanvas.width, depthCanvas.height);
   817→        return;
   818→      }
   819→      const mapEl = map.getContainer();
   820→      const w = mapEl.clientWidth;
   821→      const h = mapEl.clientHeight;
   822→      const dpr = window.devicePixelRatio || 1;
   823→      depthCanvas.width = w * dpr;
   824→      depthCanvas.height = h * dpr;
   825→      depthCanvas.style.width = w + 'px';
   826→      depthCanvas.style.height = h + 'px';
   827→
   828→      const ctx = depthCanvas.getContext('2d');
   829→      if (!ctx) return;
   830→      ctx.save();
   831→      ctx.scale(dpr, dpr);
   832→      ctx.clearRect(0, 0, w, h);
   833→
   834→      // Compute px-per-degree at the current map centre.
   835→      const center = map.getCenter();
   836→      const pt1 = map.latLngToContainerPoint(center);
   837→      const pt2Lng = map.latLngToContainerPoint({
   838→        lat: center.lat,
   839→        lng: center.lng + 1,
   840→      } as L.LatLngExpression);
   841→      const pt2Lat = map.latLngToContainerPoint({
   842→        lat: center.lat + 1,
   843→        lng: center.lng,
   844→      } as L.LatLngExpression);
   845→      const pxPerDegLng = Math.abs(pt2Lng.x - pt1.x);
   846→      const pxPerDegLat = Math.abs(pt2Lat.y - pt1.y);
   847→      // For interval picking we use the average; the grid is square in degrees
   848→      // anyway, so spacing is roughly the same on both axes near the equator.
   849→      const pxPerDeg = (pxPerDegLng + pxPerDegLat) / 2;
   850→
   851→      const source: DepthSource = activeDepthSource;
   852→      const density = depthDensityRef.current;
   853→      const interval: GridInterval = chooseGridInterval(pxPerDeg, density, source);
   854→
   855→      const bounds = map.getBounds();
   856→      const west = bounds.getWest();
   857→      const east = bounds.getEast();
   858→      const south = Math.max(-85, bounds.getSouth());
   859→      const north = Math.min(85, bounds.getNorth());
   860→
   861→      const points: GridPoint[] = enumerateGridPoints(south, north, west, east, interval.sec);
   862→
   863→      // Suppress overlay if there are too many points — show a hint instead.
   864→      const { render, hint } = shouldRenderDepth(points.length);
   865→      setDepthHint(hint);
   866→      if (!render) {
   867→        ctx.restore();
   868→        return;
   869→      }
   870→
   871→      // Update the source label so the Layers panel can show it.
   872→      // (We don't need to setState every frame — only when interval changes.)
   873→      // For simplicity we leave the source label static (it depends on env
   874→      // vars, not on zoom).
   875→
   876→      // Draw every cached ocean point as a number with white halo.
   877→      ctx.font = '11px "Courier New", ui-monospace, monospace';
   878→      ctx.textAlign = 'center';
   879→      ctx.textBaseline = 'middle';
   880→      const EDGE_MARGIN = 14; // px — skip points within 14 px of map edges
   881→      for (const p of points) {
   882→        const elev = getCachedDepth(p.lat, p.lng);
   883→        if (elev === null) continue; // not yet cached — will appear next redraw
   884→        const text = formatDepth(elev, source);
   885→        if (text === null) continue; // land — skip
   886→
   887→        const pt = map.latLngToContainerPoint({ lat: p.lat, lng: p.lng } as L.LatLngExpression);
   888→        if (pt.x < EDGE_MARGIN || pt.x > w - EDGE_MARGIN) continue;
   889→        if (pt.y < EDGE_MARGIN || pt.y > h - EDGE_MARGIN) continue;
   890→
   891→        // White halo (3 px stroke first), then dark blue-gray text on top.
   892→        ctx.lineJoin = 'round';
   893→        ctx.strokeStyle = 'rgba(255,255,255,0.95)';
   894→        ctx.lineWidth = 3;
   895→        ctx.strokeText(text, pt.x, pt.y);
   896→        ctx.fillStyle = '#1e3a5f'; // dark blue-gray, reads on road + satellite
   897→        ctx.fillText(text, pt.x, pt.y);
   898→      }
   899→      ctx.restore();
   900→    }

   902→    function scheduleDepthRedraw() {
   903→      if (depthDebounce !== null) {
   904→        clearTimeout(depthDebounce);
   905→      }
   906→      // Debounce 400 ms after the map stops moving, then redraw via RAF.
   907→      depthDebounce = setTimeout(() => {
   908→        depthDebounce = null;
   909→        if (depthRAF !== null) return;
   910→        depthRAF = requestAnimationFrame(() => {
   911→          depthRAF = null;
   912→          drawDepthOverlay();
   913→          // Kick off a fetch for any uncached points in the current viewport.
   914→          // The fetch resolves asynchronously; when it does, we redraw again
   915→          // so the new values appear.
   916→          void maybeFetchDepths();
   917→        });
   918→      }, 400);
   919→    }

   921→    /**
   922→     * Fire a depth fetch for the current viewport if the layer is enabled
   923→     * and there are uncached grid points. Supersedes any in-flight fetch via
   924→     * AbortController. On error/rate-limit, sets `depthStatus` so the UI
   925→     * can show a non-blocking notice and continues using cached values.
   926→     */
   927→    async function maybeFetchDepths(): Promise<void> {
   928→      if (!depthEnabledRef.current) return;
   929→      if (activeDepthSource === 'NONE') return;
   930→      // Abort the previous in-flight fetch.
   931→      if (depthFetchAbort) {
   932→        depthFetchAbort.abort();
   933→      }
   934→      const ctrl = new AbortController();
   935→      depthFetchAbort = ctrl;
   936→
   937→      // Re-enumerate points (cheap) — the viewport may have shifted during
   938→      // the debounce window.
   939→      const center = map.getCenter();
   940→      const pt1 = map.latLngToContainerPoint(center);
   941→      const pt2Lng = map.latLngToContainerPoint({
   942→        lat: center.lat,
   943→        lng: center.lng + 1,
   944→      } as L.LatLngExpression);
   945→      const pt2Lat = map.latLngToContainerPoint({
   946→        lat: center.lat + 1,
   947→        lng: center.lng,
   948→      } as L.LatLngExpression);
   949→      const pxPerDeg = (Math.abs(pt2Lng.x - pt1.x) + Math.abs(pt2Lat.y - pt1.y)) / 2;
   950→      const interval = chooseGridInterval(pxPerDeg, depthDensityRef.current, activeDepthSource);
   951→      const bounds = map.getBounds();
   952→      const points = enumerateGridPoints(
   953→        Math.max(-85, bounds.getSouth()),
   954→        Math.min(85, bounds.getNorth()),
   955→        bounds.getWest(),
   956→        bounds.getEast(),
   957→        interval.sec,
   958→      );
   959→      if (points.length === 0) return;
   960→      if (points.length > 400) return; // would be suppressed anyway
   961→
   962→      setDepthStatus('loading');
   963→      try {
   964→        const result = await fetchDepths(points, ctrl.signal);
   965→        if (ctrl.signal.aborted) return;
   966→        setDepthStatus(result.status);
   967→        // Force a redraw so the new values appear.
   968→        if (depthRAF !== null) return;
   969→        depthRAF = requestAnimationFrame(() => {
   970→          depthRAF = null;
   971→          drawDepthOverlay();
   972→        });
   973→      } catch {
   974→        if (!ctrl.signal.aborted) {
   975→          setDepthStatus('error');
   976→        }
   977→      }
   978→    }
   979→
