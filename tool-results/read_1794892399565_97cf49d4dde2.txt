   228→function bankBy(bank: DataBankField[], key: string): DataBankField | undefined {
   229→  return bank.find((f) => f.key === key && (f.value ?? '').trim().length > 0);
   230→}
   231→
   232→/** Nilai bank siap pakai, atau null. */
   233→function bankValue(bank: DataBankField[], key: string): string | null {
   234→  const f = bankBy(bank, key);
   235→  return f ? f.value!.trim() : null;
   236→}
   237→
   238→/** Tipe sumber asli field bank (case / manual_confirmed / actual_report), atau fallback. */
   239→function bankSrc(bank: DataBankField[], key: string, fallback: SourceType = 'case'): SourceType {
   240→  const f = bankBy(bank, key);
   241→  return f ? f.sourceType : fallback;
   242→}
   243→
   244→function coordText(p?: SnapshotLatLng | null): string | null {
   245→  if (!p || !Number.isFinite(p.lat) || !Number.isFinite(p.lng)) return null;
   246→  if (p.lat === 0 && p.lng === 0) return null;
   247→  return `${formatDMS(p.lat, false)} ${formatDMS(p.lng, true)}`;
   248→}
   249→
   250→function numText(v: number | undefined | null, digits = 1): string | null {
   251→  if (v === undefined || v === null || !Number.isFinite(v) || v === 0) return null;
   252→  return String(Number(v.toFixed(digits)));
   253→}

   255→function field(
   256→  key: string,
   257→  label: string,
   258→  value: string | null,
   259→  opts: {
   260→    unit?: string;
   261→    sourceType?: SourceType;
   262→    sourceRef?: string;
   263→    status?: FieldStatus;
   264→    plannedActual?: PlannedActual;
   265→    question?: string;
   266→  } = {},
   267→): ResolvedField {
   268→  const status = opts.status ?? (value && value.trim() ? 'READY' : 'PENDING');
   269→  return {
   270→    key,
   271→    label,
   272→    value: value && value.trim() ? value.trim() : PENDING_TEXT,
   273→    unit: opts.unit,
   274→    sourceType: opts.sourceType ?? (value && value.trim() ? 'case' : 'unknown'),
   275→    sourceRef: opts.sourceRef,
   276→    status,
   277→    plannedActual: opts.plannedActual,
   278→    question: opts.question,
   279→  };
   280→}

   282→/** Tambahkan pertanyaan default bila field PENDING tanpa pertanyaan. */
   283→function withDefaultQuestion(f: ResolvedField): ResolvedField {
   284→  if (f.status === 'PENDING' && !f.question) {
   285→    f.question = `${f.label}: mohon lengkapi datanya.`;
   286→  }
   287→  if (f.status === 'ACTUAL_REQUIRED' && !f.question) {
   288→    f.question = `${f.label}: belum ada laporan aktual — mohon isi setelah penugasan (laporan SRU/debrief).`;
   289→  }
   290→  return f;
   291→}

   293→/** Sudut polygon search area dari center + panjang + lebar + orientasi (° dari utara). */
   294→export function computeAreaCorners(
   295→  center: SnapshotLatLng,
   296→  lengthNm: number,
   297→  widthNm: number,
   298→  orientationDeg: number,
   299→): SnapshotLatLng[] {
   300→  const rad = (orientationDeg * Math.PI) / 180;
   301→  const dxL = (lengthNm / 2) * Math.sin(rad) / 60;
   302→  const dyL = (lengthNm / 2) * Math.cos(rad) / 60;
   303→  const dxW = (widthNm / 2) * Math.cos(rad) / 60;
   304→  const dyW = (widthNm / 2) * -Math.sin(rad) / 60;
   305→  return [
   306→    { lat: center.lat + dyL + dyW, lng: center.lng + dxL + dxW }, // A
   307→    { lat: center.lat + dyL - dyW, lng: center.lng + dxL - dxW }, // B
   308→    { lat: center.lat - dyL - dyW, lng: center.lng - dxL - dxW }, // C
   309→    { lat: center.lat - dyL + dyW, lng: center.lng - dxL + dxW }, // D
   310→  ];
   311→}

   320→/**
   321→ * Routing sumber Track Spacing sesuai Panduan Agent §7:
   322→ *  - Fixed wing / helikopter pencarian maritim → WS 2
   323→ *  - Kapal / RIB / Sea Rider → WS 8
   324→ *  - Land search → WS 3 (W = Wu × Fv × Ff; S khusus WS3)
   325→ *  - Pola/metode lain → data tasking/peta; agent tidak membuat nilai sendiri.
   326→ */
   327→export function resolveTrackSpacing(snap: DocumentSnapshot | null, bank: DataBankField[]): TrackSpacingResolution {
   328→  const ts = snap?.trackSpacing ?? {};
   329→  const env = snap?.pattern?.environment ?? '';
   330→  const src = snap?.pattern?.worksheetSource ?? '';
   331→  const bankTs = bankValue(bank, 'trackSpacingPlan');
   332→
   333→  // 1) Sumber eksplisit yang dipilih pengguna pada peta/worksheet.
   334→  if (src === 'ws2' && numText(ts.ws2Nm)) {
   335→    return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Fixed-wing/helikopter pencarian maritim → sumber S = WS 2.' };
   336→  }
   337→  if (src === 'ws8' && numText(ts.ws8Nm)) {
   338→    return { valueNm: ts.ws8Nm!, sourceRef: 'Worksheet 8 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Kapal/RIB/Sea Rider → sumber S = WS 8.' };
   339→  }
   340→  if (src === 'ws3' && numText(ts.ws3Nm)) {
   341→    return { valueNm: ts.ws3Nm!, sourceRef: 'Worksheet 3 — Land Search Track Spacing (S)', sourceType: 'worksheet', note: 'Pencarian darat → sumber S = WS 3.' };
   342→  }
   343→
   344→  // 2) Routing otomatis menurut lingkungan pencarian.
   345→  if (env === 'land' && numText(ts.ws3Nm)) {
   346→    return { valueNm: ts.ws3Nm!, sourceRef: 'Worksheet 3 — Land Search Track Spacing (S)', sourceType: 'worksheet', note: 'Lingkungan darat → sumber S = WS 3.' };
   347→  }
   348→  if (env === 'water') {
   349→    const hasAirAsset = (snap?.assets ?? []).some((a) => a.type === 'helicopter' || a.type === 'fixed-wing-aircraft');
   350→    if (hasAirAsset && numText(ts.ws2Nm)) {
   351→      return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Aset udara pada pencarian maritim → sumber S = WS 2.' };
   352→    }
   353→    if (numText(ts.ws8Nm)) {
   354→      return { valueNm: ts.ws8Nm!, sourceRef: 'Worksheet 8 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Aset laut → sumber S = WS 8.' };
   355→    }
   356→    if (numText(ts.ws2Nm)) {
   357→      return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'S dari WS 2 (pencarian maritim).' };
   358→    }
   359→  }
   360→
   361→  // 3) Bank data (hanya bila nilai S benar-benar tertulis di soal/rencana).
   362→  if (bankTs) {
   363→    return { valueNm: Number(bankTs) || null, sourceRef: 'Bank Data (soal kasus)', sourceType: 'case', note: 'S dari Soal Kasus — pastikan sesuai worksheet routing.' };
   364→  }
   365→  return { valueNm: null, sourceRef: '', sourceType: 'unknown', note: 'S tidak diisi dari pola visual — jalankan worksheet routing yang sesuai (WS 2 / WS 3 / WS 8).' };
   366→}

   394→/** Lokasi kejadian: marker LKP peta/worksheet → bank → PENDING. */
   395→function resolveLkp(snap: DocumentSnapshot | null, bank: DataBankField[]): SourcePick & { conflict?: string } {
   396→  const mapCoord = snap?.lkpSet ? coordText(snap.lkp) : null;
   397→  const bankCoord = bankValue(bank, 'lkpCoord');
   398→  if (mapCoord && bankCoord && mapCoord !== bankCoord) {
   399→    // Tampil keduanya — konflik minta verifikasi SMC (Panduan Agent §15).
   400→    return {
   401→      value: `${mapCoord} (peta/worksheet) ‖ ${bankCoord} (soal kasus)`,
   402→      sourceType: 'map',
   403→      sourceRef: 'LKP marker peta vs Soal Kasus',
   404→      conflict: `LKP peta (${mapCoord}) berbeda dengan Soal Kasus (${bankCoord}) — minta verifikasi SMC.`,
   405→    };
   406→  }
   407→  if (mapCoord) return { value: mapCoord, sourceType: 'map', sourceRef: 'LKP marker peta aplikasi' };
   408→  if (bankCoord) return { value: bankCoord, sourceType: 'case', sourceRef: 'Soal Kasus' };
   409→  return { value: null, sourceType: 'unknown' };
   410→}

   412→/** Posisi datum: HANYA dari hasil worksheet (WS 1 / Datum). */
   413→function resolveDatumText(snap: DocumentSnapshot | null): SourcePick {
   414→  const d = snap?.datum;
   415→  const left = coordText(d?.datumLeft ?? null);
   416→  const right = coordText(d?.datumRight ?? null);
   417→  if (left && right) {
   418→    return { value: `Datum Kiri: ${left} — Datum Kanan: ${right}`, sourceType: 'worksheet', sourceRef: 'Worksheet 1 (Datum)' };
   419→  }
   420→  if (left) return { value: `Datum: ${left}`, sourceType: 'worksheet', sourceRef: 'Worksheet 1 (Datum)' };
   421→  return { value: null, sourceType: 'unknown' };
   422→}

   424→/** Koordinat sudut Search Area (A–D): dari hasil worksheet/geometry peta. */
   425→function resolveAreaCorners(snap: DocumentSnapshot | null): SourcePick {
   426→  const sa = snap?.searchArea;
   427→  if (sa?.center && sa.lengthNm && sa.widthNm) {
   428→    const corners = computeAreaCorners(sa.center, sa.lengthNm, sa.widthNm, sa.orientationDeg ?? 0);
   429→    const labels = ['A', 'B', 'C', 'D'];
   430→    const text = corners
   431→      .map((c, i) => `${labels[i]}: ${coordText(c)}`)
   432→      .join('\n');
   433→    return { value: text, sourceType: 'worksheet', sourceRef: 'Worksheet 1 / geometri Search Area peta' };
   434→  }

   449→  const opDay = bankValue(bank, 'opDay') ?? '...';

   469→    return { title: 'BRIEFING UNIT PENCARIAN DAN PERTOLONGAN', subtitle: 'Format B — Lampiran II Perban 3/2020' };
   470→  }
   471→  return { title: 'DE-BRIEFING UNIT PENCARIAN DAN PERTOLONGAN', subtitle: 'Format C — Lampiran II Perban 3/2020' };

   499→function buildAllocationTables(snap: DocumentSnapshot | null, bank: DataBankField[]): DocTable[] {
   500→  const assets = snap?.assets ?? [];
   501→  const airAssets = assets.filter((a) => a.type === 'helicopter' || a.type === 'fixed-wing-aircraft');
   502→  const seaAssets = assets.filter((a) => ['rescue-vessel', 'patrol-boat', 'rib', 'sea-rider'].includes(a.type));
   503→  const landAssets = assets.filter((a) => !airAssets.includes(a) && !seaAssets.includes(a));
   504→  const bankAir = bankValue(bank, 'preparedSruAir');
   505→  const bankSea = bankValue(bank, 'preparedSruSea');
   506→  const bankLand = bankValue(bank, 'preparedSruLand');
   507→  const tables: DocTable[] = [];
   508→
   509→  const mkRow = (a: DocumentSnapshot['assets'][number] | null, name: string): string[] => {
   510→    if (!a) return [name, '-', '-', '-', '-', PENDING_TEXT, PENDING_TEXT];
   511→    const endurance = numText(a.enduranceHours, 1) ? `${numText(a.enduranceHours, 1)} jam` : PENDING_TEXT;
   512→    const transit = numText(a.transitSpeedKts, 0) ? `${numText(a.transitSpeedKts, 0)} kts` : PENDING_TEXT;
   513→    return [name, endurance, transit, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT];
   514→  };
   515→  const cols = ['Assets', 'Total Endurance', 'Transit', 'Reserve', 'Weather (TAF)', 'OSE', 'Keterangan'];
   516→  const prov: string[] = [];
   517→  const collect = (list: DocumentSnapshot['assets'], fallback: string | null, domain: string) => {
   518→    const rows: string[][] = [];
   519→    if (list.length === 0 && fallback) {
   520→      rows.push([fallback, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT]);
   521→      prov.push(`${domain}: ${fallback} — sumber Soal Kasus (detail endurance/transit perlu data aset).`);
   522→    } else if (list.length === 0) {
   523→      rows.push(['—', '-', '-', '-', '-', '-', '-']);
   524→    } else {
   525→      for (const a of list) {
   526→        rows.push(mkRow(a, a.name + (a.assignedSubArea ? ` (${a.assignedSubArea})` : '')));
   527→        prov.push(`${a.name}: endurance/speed dari Alokasi Aset SRU (worksheet); Reserve/OSE/TAF belum tersedia.`);
   528→      }
   529→    }
   530→    return rows;
   531→  };
   532→
   533→  tables.push({ title: '1) Udara', columns: cols, rows: collect(airAssets, bankAir, 'SRU udara'), provenance: prov.slice() });
   534→  tables.push({ title: '2) Laut', columns: cols, rows: collect(seaAssets, bankSea, 'SRU laut'), provenance: [] });
   535→  tables.push({ title: '3) Darat', columns: cols, rows: collect(landAssets, bankLand, 'SRU darat'), provenance: [] });
   536→  return tables;
   537→}

   894→function validateDocument(sections: DocSection[], conflictWarnings: string[]) {
   895→  let ready = 0; let pending = 0; let conflict = conflictWarnings.length; let actualRequired = 0; let total = 0;
   896→  const counts: Record<SourceType, number> = {
   897→    case: 0, worksheet: 0, map: 0, asset: 0, personnel: 0,
   898→    communications: 0, actual_report: 0, manual_confirmed: 0, unknown: 0,
   899→  };
   900→  for (const s of sections) {
   901→    for (const f of s.fields ?? []) {
   902→      total += 1;
   903→      if (f.status === 'READY') ready += 1;
   904→      else if (f.status === 'PENDING') pending += 1;
   905→      else if (f.status === 'CONFLICT') conflict += 1;
   906→      else if (f.status === 'ACTUAL_REQUIRED') actualRequired += 1;
   907→      counts[f.sourceType] = (counts[f.sourceType] ?? 0) + 1;
   908→    }
   909→  }
   910→  return { ready, pending, conflict, actualRequired, total, conflictWarnings, counts: counts as Record<string, number> };
   911→}
   912→
   913→function collectQuestions(sections: DocSection[]): { key: string; label: string; question: string }[] {
   914→  const out: { key: string; label: string; question: string }[] = [];
   915→  for (const s of sections) {
   916→    for (const f of s.fields ?? []) {
   917→      if ((f.status === 'PENDING' || f.status === 'ACTUAL_REQUIRED') && f.question) {
   918→        out.push({ key: f.key, label: f.label, question: f.question });
   919→      }
   920→    }
   921→  }
   922→  return out;
   923→}
   924→
   925→function countSources(sections: DocSection[]): Record<string, number> {
   926→  const counts: Record<string, number> = {};
   927→  for (const s of sections) for (const f of s.fields ?? []) counts[f.sourceType] = (counts[f.sourceType] ?? 0) + 1;
   928→  return counts;
   929→}

   947→/** Pilih format RENOPS sesuai incident type (Panduan Agent §3). */
   948→export function pickRenopsType(bank: DataBankField[]): DocType {
   949→  const t = (bankValue(bank, 'incidentLabel') ?? '').toLowerCase();
   950→  const disasterKeys = ['bencana', 'membahayakan', 'tenggelam', 'hilang', 'kondisi membahayakan'];
   951→  const accidentKeys = ['kecelakaan', 'crash', 'accident', 'kapal', 'pesawat'];
   952→  if (disasterKeys.some((k) => t.includes(k)) && !t.includes('kecelakaan kapal') && !t.includes('kecelakaan pesawat')) return 'RENOPS_II';
   953→  if (accidentKeys.some((k) => t.includes(k))) return 'RENOPS_I';
   954→  return 'RENOPS_II';
   955→}
