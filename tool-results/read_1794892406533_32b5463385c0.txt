   244→   244→function coordText(p?: SnapshotLatLng | null): string | null {
   245→   245→  if (!p || !Number.isFinite(p.lat) || !Number.isFinite(p.lng)) return null;
   246→   246→  if (p.lat === 0 && p.lng === 0) return null;
   247→   247→  return `${formatDMS(p.lat, false)} ${formatDMS(p.lng, true)}`;
   248→   248→}
   249→   249→
   250→   250→function numText(v: number | undefined | null, digits = 1): string | null {
   251→   251→  if (v === undefined || v === null || !Number.isFinite(v) || v === 0) return null;
   252→   252→  return String(Number(v.toFixed(digits)));
   253→   253→}

   293→   293→/** Sudut polygon search area dari center + panjang + lebar + orientasi (° dari utara). */
   294→   294→export function computeAreaCorners(
   295→   295→  center: SnapshotLatLng,
   296→   296→  lengthNm: number,
   297→   297→  widthNm: number,
   298→   298→  orientationDeg: number,
   299→   299→): SnapshotLatLng[] {
   300→   300→  const rad = (orientationDeg * Math.PI) / 180;
   301→   301→  const dxL = (lengthNm / 2) * Math.sin(rad) / 60;
   302→   302→  const dyL = (lengthNm / 2) * Math.cos(rad) / 60;
   303→   303→  const dxW = (widthNm / 2) * Math.cos(rad) / 60;
   304→   304→  const dyW = (widthNm / 2) * -Math.sin(rad) / 60;
   305→   305→  return [
   306→   306→    { lat: center.lat + dyL + dyW, lng: center.lng + dxL + dxW }, // A
   307→   307→    { lat: center.lat + dyL - dyW, lng: center.lng + dxL - dxW }, // B
   308→   308→    { lat: center.lat - dyL - dyW, lng: center.lng - dxL - dxW }, // C
   309→   309→    { lat: center.lat - dyL + dyW, lng: center.lng - dxL + dxW }, // D
   310→   310→  ];
   311→   311→}

   314→   314→  valueNm: number | null;

   320→   320→/**
   321→   321→ * Routing sumber Track Spacing sesuai Panduan Agent §7:
   322→   322→ *  - Fixed wing / helikopter pencarian maritim → WS 2
   323→   323→ *  - Kapal / RIB / Sea Rider → WS 8
   324→   324→ *  - Land search → WS 3 (W = Wu × Fv × Ff; S khusus WS3)
   325→   325→ *  - Pola/metode lain → data tasking/peta; agent tidak membuat nilai sendiri.
   326→   326→ */
   327→   327→export function resolveTrackSpacing(snap: DocumentSnapshot | null, bank: DataBankField[]): TrackSpacingResolution {
   328→   328→  const ts = snap?.trackSpacing ?? {};
   329→   329→  const env = snap?.pattern?.environment ?? '';
   330→   330→  const src = snap?.pattern?.worksheetSource ?? '';
   331→   331→  const bankTs = bankValue(bank, 'trackSpacingPlan');
   332→   332→
   333→   333→  // 1) Sumber eksplisit yang dipilih pengguna pada peta/worksheet.
   334→   334→  if (src === 'ws2' && numText(ts.ws2Nm)) {
   335→   335→    return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Fixed-wing/helikopter pencarian maritim → sumber S = WS 2.' };
   336→   336→  }
   337→   337→  if (src === 'ws8' && numText(ts.ws8Nm)) {
   338→   338→    return { valueNm: ts.ws8Nm!, sourceRef: 'Worksheet 8 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Kapal/RIB/Sea Rider → sumber S = WS 8.' };
   339→   339→  }
   340→   340→  if (src === 'ws3' && numText(ts.ws3Nm)) {
   341→   341→    return { valueNm: ts.ws3Nm!, sourceRef: 'Worksheet 3 — Land Search Track Spacing (S)', sourceType: 'worksheet', note: 'Pencarian darat → sumber S = WS 3.' };
   342→   342→  }
   343→   343→
   344→   344→  // 2) Routing otomatis menurut lingkungan pencarian.
   345→   345→  if (env === 'land' && numText(ts.ws3Nm)) {
   346→   346→    return { valueNm: ts.ws3Nm!, sourceRef: 'Worksheet 3 — Land Search Track Spacing (S)', sourceType: 'worksheet', note: 'Lingkungan darat → sumber S = WS 3.' };
   347→   347→  }
   348→   348→  if (env === 'water') {
   349→   349→    const hasAirAsset = (snap?.assets ?? []).some((a) => a.type === 'helicopter' || a.type === 'fixed-wing-aircraft');
   350→   350→    if (hasAirAsset && numText(ts.ws2Nm)) {
   351→   351→      return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Aset udara pada pencarian maritim → sumber S = WS 2.' };
   352→   352→    }
   353→   353→    if (numText(ts.ws8Nm)) {
   354→   354→      return { valueNm: ts.ws8Nm!, sourceRef: 'Worksheet 8 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'Aset laut → sumber S = WS 8.' };
   355→   355→    }
   356→   356→    if (numText(ts.ws2Nm)) {
   357→   357→      return { valueNm: ts.ws2Nm!, sourceRef: 'Worksheet 2 — Practical Track Spacing (S)', sourceType: 'worksheet', note: 'S dari WS 2 (pencarian maritim).' };
   358→   358→    }
   359→   359→  }
   360→   360→
   361→   361→  // 3) Bank data (hanya bila nilai S benar-benar tertulis di soal/rencana).
   362→   362→  if (bankTs) {
   363→   363→    return { valueNm: Number(bankTs) || null, sourceRef: 'Bank Data (soal kasus)', sourceType: 'case', note: 'S dari Soal Kasus — pastikan sesuai worksheet routing.' };
   364→   364→  }
   365→   365→  return { valueNm: null, sourceRef: '', sourceType: 'unknown', note: 'S tidak diisi dari pola visual — jalankan worksheet routing yang sesuai (WS 2 / WS 3 / WS 8).' };
   366→   366→}

   395→   395→function resolveLkp(snap: DocumentSnapshot | null, bank: DataBankField[]): SourcePick & { conflict?: string } {
   396→   396→  const mapCoord = snap?.lkpSet ? coordText(snap.lkp) : null;
   397→   397→  const bankCoord = bankValue(bank, 'lkpCoord');
   398→   398→  if (mapCoord && bankCoord && mapCoord !== bankCoord) {
   399→   399→    // Tampil keduanya — konflik minta verifikasi SMC (Panduan Agent §15).
   400→   400→    return {
   401→   401→      value: `${mapCoord} (peta/worksheet) ‖ ${bankCoord} (soal kasus)`,
   402→   402→      sourceType: 'map',
   403→   403→      sourceRef: 'LKP marker peta vs Soal Kasus',
   404→   404→      conflict: `LKP peta (${mapCoord}) berbeda dengan Soal Kasus (${bankCoord}) — minta verifikasi SMC.`,
   405→   405→    };
   406→   406→  }
   407→   407→  if (mapCoord) return { value: mapCoord, sourceType: 'map', sourceRef: 'LKP marker peta aplikasi' };
   408→   408→  if (bankCoord) return { value: bankCoord, sourceType: 'case', sourceRef: 'Soal Kasus' };
   409→   409→  return { value: null, sourceType: 'unknown' };
   410→   410→}

   413→   413→function resolveDatumText(snap: DocumentSnapshot | null): SourcePick {

   424→   424→/** Koordinat sudut Search Area (A–D): dari hasil worksheet/geometry peta. */
   425→   425→function resolveAreaCorners(snap: DocumentSnapshot | null): SourcePick {
   426→   426→  const sa = snap?.searchArea;
   427→   427→  if (sa?.center && sa.lengthNm && sa.widthNm) {
   428→   428→    const corners = computeAreaCorners(sa.center, sa.lengthNm, sa.widthNm, sa.orientationDeg ?? 0);
   429→   429→    const labels = ['A', 'B', 'C', 'D'];
   430→   430→    const text = corners
   431→   431→      .map((c, i) => `${labels[i]}: ${coordText(c)}`)
   432→   432→      .join('\n');
   433→   433→    return { value: text, sourceType: 'worksheet', sourceRef: 'Worksheet 1 / geometri Search Area peta' };
   434→   434→  }
   435→   435→  // Fallback: area dari datum kiri/kanan (area dat-2 titik) — tetap sumber worksheet.
   436→   436→  const d = snap?.datum;
   437→   437→  const left = coordText(d?.datumLeft ?? null);
   438→   438→  const right = coordText(d?.datumRight ?? null);
   439→   439→  if (left && right) {
   440→   440→    return { value: `A: ${left}\nD: ${right}`, sourceType: 'worksheet', sourceRef: 'Worksheet 1 (Datum kiri/kanan)' };
   441→   441→  }
   442→   442→  const bankArea = null; // koordinat area BUKAN fakta soal — wajib hasil worksheet/peta.
   443→   443→  void bankArea;
   444→   444→  return { value: null, sourceType: 'unknown' };
   445→   445→}

   499→   499→function buildAllocationTables(snap: DocumentSnapshot | null, bank: DataBankField[]): DocTable[] {
   500→   500→  const assets = snap?.assets ?? [];
   501→   501→  const airAssets = assets.filter((a) => a.type === 'helicopter' || a.type === 'fixed-wing-aircraft');
   502→   502→  const seaAssets = assets.filter((a) => ['rescue-vessel', 'patrol-boat', 'rib', 'sea-rider'].includes(a.type));
   503→   503→  const landAssets = assets.filter((a) => !airAssets.includes(a) && !seaAssets.includes(a));
   504→   504→  const bankAir = bankValue(bank, 'preparedSruAir');
   505→   505→  const bankSea = bankValue(bank, 'preparedSruSea');
   506→   506→  const bankLand = bankValue(bank, 'preparedSruLand');
   507→   507→  const tables: DocTable[] = [];
   508→   508→
   509→   509→  const mkRow = (a: DocumentSnapshot['assets'][number] | null, name: string): string[] => {
   510→   510→    if (!a) return [name, '-', '-', '-', '-', PENDING_TEXT, PENDING_TEXT];
   511→   511→    const endurance = numText(a.enduranceHours, 1) ? `${numText(a.enduranceHours, 1)} jam` : PENDING_TEXT;
   512→   512→    const transit = numText(a.transitSpeedKts, 0) ? `${numText(a.transitSpeedKts, 0)} kts` : PENDING_TEXT;
   513→   513→    return [name, endurance, transit, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT];
   514→   514→  };
   515→   515→  const cols = ['Assets', 'Total Endurance', 'Transit', 'Reserve', 'Weather (TAF)', 'OSE', 'Keterangan'];
   516→   516→  const prov: string[] = [];
   517→   517→  const collect = (list: DocumentSnapshot['assets'], fallback: string | null, domain: string) => {
   518→   518→    const rows: string[][] = [];
   519→   519→    if (list.length === 0 && fallback) {
   520→   520→      rows.push([fallback, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT, PENDING_TEXT]);
   521→   521→      prov.push(`${domain}: ${fallback} — sumber Soal Kasus (detail endurance/transit perlu data aset).`);
   522→   522→    } else if (list.length === 0) {
   523→   523→      rows.push(['—', '-', '-', '-', '-', '-', '-']);
   524→   524→    } else {
   525→   525→      for (const a of list) {
   526→   526→        rows.push(mkRow(a, a.name + (a.assignedSubArea ? ` (${a.assignedSubArea})` : '')));
   527→   527→        prov.push(`${a.name}: endurance/speed dari Alokasi Aset SRU (worksheet); Reserve/OSE/TAF belum tersedia.`);
   528→   528→      }
   529→   529→    }
   530→   530→    return rows;
   531→   531→  };
   532→   532→
   533→   533→  tables.push({ title: '1) Udara', columns: cols, rows: collect(airAssets, bankAir, 'SRU udara'), provenance: prov.slice() });
   534→   534→  tables.push({ title: '2) Laut', columns: cols, rows: collect(seaAssets, bankSea, 'SRU laut'), provenance: [] });
   535→   535→  tables.push({ title: '3) Darat', columns: cols, rows: collect(landAssets, bankLand, 'SRU darat'), provenance: [] });
   536→   536→  return tables;
   537→   537→}
